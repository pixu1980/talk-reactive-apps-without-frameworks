Step 3 - `AttributePart` turns model into DOM listener

Template engine sees `model=${"..."}`. `AttributePart.setValue()` detects the directive and calls `commitModel()`.

Route model directive


setValue(value) {
  if (this.name === "model" && isDirective(value, "model")) {
    this.commitModel(value.payload)
    this.value = value

    return
  }

  if (isSignalLike(value)) {
    this.disposeModel()
    this.bindSignal(value, (resolved) => this.commit(resolved))

    return
  }

  this.disposeModel()
  this.disposeSignal()
  this.commit(value)
}

Browser listener


const onInput = (event) => {
  const target = event.currentTarget
  const nextValue = property === "checked" ? target.checked : target[property]

  binding.config.set(nextValue)
}
